#include <stdio.h>

int main(void)
{
    double price, gross, discount, gst, total, net;
    int qty, senior;

    printf("Enter price, quantity and senior status (0/1): ");
    if (scanf("%lf %d %d", &price, &qty, &senior) != 3 ||
        price < 0 || qty < 0 || (senior != 0 && senior != 1)) {
        printf("Invalid input\n");
        return 1;
    }

    gross = price * qty;
    discount = gross * 0.15 * senior;
    gst = gross * 0.05;          /* matches the supplied expression */
    total = gross - discount + gst;
    net = (total > 1000.0) ? total - 50.0 : total;

    printf("Gross amount : Rs %.2f\n", gross);
    printf("Discount     : Rs %.2f\n", discount);
    printf("GST          : Rs %.2f\n", gst);
    printf("Total        : Rs %.2f\n", total);
    printf("Net payable  : Rs %.2f\n", net);
    return 0;
}
